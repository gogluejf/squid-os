Create Environment loader

Environment Domain

Implement the full environment loader that assembles all sections and returns a formatted string for the sys1 message.

Files

Data Shapes

Contract

func LoadEnvironment(paths config.Paths, settings config.Settings, homeDir string, workingDir string) Environment func FormatEnvironment(env Environment) string

Code Example
    func LoadEnvironment(paths config.Paths, settings config.Settings, homeDir string, workingDir string) Environment {
        env := Environment{
            OS:       CollectOSInfo(homeDir),
            Skills:   loadSkillEntries(paths.Skills),
            SquidOS: SquidOSInfo{
                SkillsDir:     paths.Skills,
                LogsDir:       paths.Logs,
                SysPromptsDir: paths.SysPrompts,
                SessionsDir:   paths.Sessions,
                DebugEnabled:  settings.DebugEnabled,
            },
            Project: loadProjectInfo(paths.ProjectDir, workingDir),
        }
        return env
    }

    func FormatEnvironment(env Environment) string {
        var b strings.Builder
        b.WriteString("# Environment

")

        // [OS] section
        b.WriteString("## [OS]
")
        b.WriteString(fmt.Sprintf("OS: %s
", env.OS.OS))
        b.WriteString(fmt.Sprintf("Arch: %s
", env.OS.Arch))
        b.WriteString(fmt.Sprintf("Home: %s
", env.OS.Home))
        b.WriteString(fmt.Sprintf("Git: %s
", boolOrNot(env.OS.GitInstalled)))
        b.WriteString(fmt.Sprintf("Tree: %s
", boolOrNot(env.OS.TreeInstalled)))
        b.WriteString("
")

        // [Skill] section
        b.WriteString("## [Skill]
")
        for _, s := range env.Skills {
            b.WriteString(fmt.Sprintf("- %s: %s
", s.Name, s.Description))
        }
        if len(env.Skills) == 0 {
            b.WriteString("No skills loaded
")
        }
        b.WriteString("
")

        // [Squid-OS] section
        b.WriteString("## [Squid-OS]
")
        b.WriteString(fmt.Sprintf("Skills: %s
", env.SquidOS.SkillsDir))
        b.WriteString(fmt.Sprintf("Logs: %s
", env.SquidOS.LogsDir))
        b.WriteString(fmt.Sprintf("System Prompts: %s
", env.SquidOS.SysPromptsDir))
        b.WriteString(fmt.Sprintf("Sessions: %s
", env.SquidOS.SessionsDir))
        if env.SquidOS.DebugEnabled {
            b.WriteString("Debug: enabled
")
        }
        b.WriteString("
")

        // [Project] section
        if env.Project != nil {
            b.WriteString("## [Project]
")
            b.WriteString(fmt.Sprintf("Working Directory: %s
", env.Project.Path))
            b.WriteString(fmt.Sprintf("Under Project Dir: %s
", boolOrNot(env.Project.IsUnderProjectDir)))
            b.WriteString(fmt.Sprintf("Git Init: %s
", boolOrNot(env.Project.IsGitRepo)))
            if len(env.Project.Projects) > 0 {
                b.WriteString("Projects:
")
                for _, p := range env.Project.Projects {
                    b.WriteString(fmt.Sprintf("- %s (%s)
", p.Name, p.Path))
                }
            }
            if env.Project.FileTree != "" {
                b.WriteString("File Tree:
")
                b.WriteString("```
")
                b.WriteString(env.Project.FileTree)
                b.WriteString("```
")
            }
        }

        return b.String()
    }
