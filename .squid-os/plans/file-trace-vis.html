func Diff(oldContent, newContent string) string

Code Example
package diff

import (
	"fmt"
	"strings"
)

// Diff returns a side-by-side diff of oldContent vs newContent.
// Left column shows old lines (- prefix for removed), right shows new (+ for added).
// Unchanged lines appear in both columns with space prefix.
// Returns "" if content is identical.
func Diff(oldContent, newContent string) string {
	if oldContent == newContent {
		return ""
	}
	oldLines := strings.Split(oldContent, "\n")
	newLines := strings.Split(newContent, "\n")
	return formatSideBySide(oldLines, newLines)
}

func formatSideBySide(oldLines, newLines []string) string {
	// 1. Compute LCS via dynamic programming
	matrix := lcsMatrix(oldLines, newLines)
	// 2. Backtrack to produce edit operations
	ops := backtrack(matrix, oldLines, newLines)
	// 3. Render side-by-side columns
	return renderOps(ops, oldLines, newLines)
}

// lcsMatrix builds the DP table for LCS.
func lcsMatrix(a, b []string) [][]int {
	m, n := len(a), len(b)
	matrix := make([][]int, m+1)
	for i := 0; i <= m; i++ {
		matrix[i] = make([]int, n+1)
	}
	for i := 1; i <= m; i++ {
		for j := 1; j <= n; j++ {
			if a[i-1] == b[j-1] {
				matrix[i][j] = matrix[i-1][j-1] + 1
			} else if matrix[i-1][j] > matrix[i][j-1] {
				matrix[i][j] = matrix[i-1][j]
			} else {
				matrix[i][j] = matrix[i][j-1]
			}
		}
	}
	return matrix
}

type opType int
const (
	opEqual  opType = iota
	opRemove          // in old only
	opAdd             // in new only
)

type editOp struct {
	typ  opType
	i, j int // indices into old/new lines
}

func backtrack(matrix [][]int, a, b []string) []editOp {
	var ops []editOp
	i, j := len(a), len(b)
	for i > 0 || j > 0 {
		if i > 0 && j > 0 && a[i-1] == b[j-1] {
			ops = append(ops, editOp{opEqual, i - 1, j - 1})
			i--; j--
		} else if j > 0 && (i == 0 || matrix[i][j-1] >= matrix[i-1][j]) {
			ops = append(ops, editOp{opAdd, i, j - 1})
			j--
		} else {
			ops = append(ops, editOp{opRemove, i - 1, j})
			i--
		}
	}
	// Reverse to get forward order
	for l, r := 0, len(ops)-1; l < r; l, r = l+1, r-1 {
		ops[l], ops[r] = ops[r], ops[l]
	}
	return ops
}

func renderOps(ops []editOp, oldLines, newLines []string) string {
	// Collect aligned pairs (left, right) for each row
	type row struct{ left, right string }
	var rows []row
	for _, op := range ops {
		switch op.typ {
		case opEqual:
			rows = append(rows, row{"  " + oldLines[op.i], "  " + newLines[op.j]})
		case opRemove:
			rows = append(rows, row{"- " + oldLines[op.i], ""})
		case opAdd:
			rows = append(rows, row{"", "+ " + newLines[op.j]})
		}
	}
	// Determine max left column width
	maxLeft := 20
	for _, r := range rows {
		if w := len(r.left); w > maxLeft {
			maxLeft = w
		}
	}
	// Format with padding
	var b strings.Builder
	for _, r := range rows {
		left := r.left
		for len(left) < maxLeft {
			left += " "
		}
		b.WriteString(fmt.Sprintf("%s │ %s\n", left, r.right))
	}
	return strings.TrimRight(b.String(), "\n")
}

  • Implement LCS-based line diff with backtrack
  • formatSideBySide produces aligned left/right columns with │ separator
  • "-" prefix on left for removed, "+" on right for added, " " for unchanged
  • Return "" if content identical

package tools

import (
	"fmt"
	"os"
	"squid-os/internal/config"
	"squid-os/internal/util"
	"sync"
)

type FileTracker struct {
	mu        sync.Mutex
	checksums map[string]string
}

func NewFileTracker() *FileTracker {
	return &FileTracker{
		checksums: make(map[string]string),
	}
}

func (ft *FileTracker) Track(entry config.FileEntry) {
	ft.mu.Lock()
	defer ft.mu.Unlock()
	ft.checksums[entry.Path] = entry.Checksum
}

func (ft *FileTracker) Validate(path string) error {
	ft.mu.Lock()
	defer ft.mu.Unlock()
	stored, ok := ft.checksums[path]
	if !ok {
		return nil // not tracked, no conflict
	}
	data, err := os.ReadFile(path)
	if err != nil {
		return fmt.Errorf("cannot read %s for validation: %w", path, err)
	}
	current := util.ComputeChecksum(data)
	if current != stored {
		return fmt.Errorf("file on disk changed: %s", path)
	}
	return nil
}

func renderFileList(files []config.FileEntry, s style.StyleLabel) string {
	if len(files) == 0 {
		return ""
	}
	var b strings.Builder
	for _, f := range files {
		checksum := f.Checksum
		if len(checksum) > 10 {
			checksum = checksum[:10]
		}
		line := fmt.Sprintf("%s (%s) %s", f.Path, f.Trace, checksum)
		b.WriteString(s.Content.Render(line) + "\n")
	}
	return strings.TrimRight(b.String(), "\n")
}
